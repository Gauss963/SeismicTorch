cmake_minimum_required(VERSION 3.20)
project(SeismicTorch VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(SEISMICTORCH_STATIC "Link a fully static executable on Linux (musl recommended)" OFF)
include(CTest)
find_package(Threads REQUIRED)
if(WIN32)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()

set(MSEED_SOURCES fileutils.c genutils.c msio.c lookup.c yyjson.c msrutils.c extraheaders.c
    pack.c packdata.c tracelist.c gmtime64.c crc32c.c parseutils.c unpack.c unpackdata.c selection.c logging.c)
list(TRANSFORM MSEED_SOURCES PREPEND "vendor/libmseed/")
add_library(mseed STATIC ${MSEED_SOURCES})
target_include_directories(mseed PUBLIC vendor/libmseed)
if(WIN32)
  target_compile_definitions(mseed PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(mseed PUBLIC ws2_32)
else()
  target_link_libraries(mseed PUBLIC m)
endif()

add_library(seismictorch STATIC src/model.cc src/inference.cc src/readers.cc)
add_library(SeismicTorch::seismictorch ALIAS seismictorch)
target_include_directories(seismictorch PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_include_directories(seismictorch PRIVATE generated)
target_link_libraries(seismictorch PRIVATE mseed Threads::Threads)
add_executable(seismictorch-infer src/main.cc)
target_link_libraries(seismictorch-infer PRIVATE seismictorch)
if(MSVC)
  target_compile_options(seismictorch PRIVATE /W4 /fp:precise)
  target_compile_options(seismictorch-infer PRIVATE /W4 /utf-8)
else()
  target_compile_options(seismictorch PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  target_compile_options(seismictorch-infer PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(SEISMICTORCH_STATIC)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "SEISMICTORCH_STATIC requires a Linux toolchain")
  endif()
  target_link_options(seismictorch-infer PRIVATE -static -static-libgcc -static-libstdc++)
endif()

if(BUILD_TESTING)
  add_executable(seismictorch-tests tests/tests.cc)
  target_link_libraries(seismictorch-tests PRIVATE seismictorch mseed)
  if(MSVC)
    target_compile_options(seismictorch-tests PRIVATE /utf-8)
  endif()
  target_compile_definitions(seismictorch-tests PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  if(SEISMICTORCH_STATIC)
    target_link_options(seismictorch-tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_executable(seismictorch-parity-probe tests/parity_probe.cc)
  target_link_libraries(seismictorch-parity-probe PRIVATE seismictorch)
  if(SEISMICTORCH_STATIC)
    target_link_options(seismictorch-parity-probe PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME native-parity-and-readers COMMAND seismictorch-tests)
  add_test(NAME cli-help COMMAND seismictorch-infer --help)
  add_test(NAME cli-model-version COMMAND seismictorch-infer --version)
endif()

install(TARGETS seismictorch-infer RUNTIME DESTINATION bin)
install(FILES include/seismictorch/inference.hh DESTINATION include/seismictorch)
